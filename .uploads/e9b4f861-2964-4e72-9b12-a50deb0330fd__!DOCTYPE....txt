// 从服务器加载数据
function loadFromServer(callback) {
  try {
    var xhr = new XMLHttpRequest();
    xhr.open('GET', '/api/data', true);
    xhr.timeout = 3000;
    xhr.onload = function() {
      if (xhr.status === 200) {
        var data = JSON.parse(xhr.responseText);
        if (data && data.users) {
          db = data;
          // 同步到localStorage作为离线备份
          localStorage.setItem(STORAGE_KEY, JSON.stringify(db));
          _lastSyncTime = data._lastUpdate || 0;
          if (callback) callback(true);
          return;
        }
      }
      if (callback) callback(false);
    };
    xhr.onerror = function() { if (callback) callback(false); };
    xhr.ontimeout = function() { if (callback) callback(false); };
    xhr.send();
  } catch (e) {
    if (callback) callback(false);
  }
}

// 保存数据 - 同时保存到服务器和localStorage
function saveData() {
  // 记录更新时间
  if (db) {
    db._lastUpdate = Date.now();
  }
  // localStorage作为本地备份
  localStorage.setItem(STORAGE_KEY, JSON.stringify(db));
  // 异步同步到服务器
  saveToServer();
}

// 保存数据到服务器
function saveToServer() {
  if (_syncing) return; // 防止重复提交
  _syncing = true;
  try {
    var xhr = new XMLHttpRequest();
    xhr.open('POST', '/api/data', true);
    xhr.setRequestHeader('Content-Type', 'application/json');
    xhr.onload = function() {
      _syncing = false;
      if (xhr.status === 200) {
        _lastSyncTime = Date.now();
      }
    };
    xhr.onerror = function() { _syncing = false; };
    xhr.send(JSON.stringify(db));
  } catch (e) {
    _syncing = false;
  }
}

// 定期从服务器拉取最新数据（每30秒）

      });
    }
  }, 30000);
}

// ==================== 库存计算 ====================
function getStock(productId, warehouseId) {
  let total = 0;
  db.stockFlow.forEach(f => {
    if (f.productId === productId && f.warehouseId === warehouseId) {
      if (f.type === 'in') total += f.qty;
      else if (f.type === 'out') total -= f.qty;
    }
  });
  return total;
}

function getProductStockAllWarehouses(productId) {
  const result = {};
  db.warehouses.forEach(w => {
    result[w.id] = getStock(productId, w.id);
  });
  return result;
}

function getWarehouseStock(warehouseId) {
  const result = {};
  db.products.forEach(p => {
    const qty = getStock(p.id, warehouseId);
    if (qty !== 0) result[p.id] = qty;
  });
  return result;
}

function getTotalStock() {
  let total = 0;
  db.stockFlow.forEach(f => {
    if (f.type === 'in') total += f.qty;
    else if (f.type === 'out') total -= f.qty;
  });
  return total;
}

// 装箱显示
function formatPack(qty, perBox) {
  if (!perBox || perBox <= 0) return qty + '台';
  const boxes = Math.floor(qty / perBox);
  const remain = qty % perBox;
  if (boxes === 0) return remain + '台';
  if (remain === 0) return boxes + '箱';
  return boxes + '箱+' + remain + '台';
}

function formatTime(ts) {
  const d = new Date(ts);
  const pad = n => n.toString().padStart(2, '0');
  return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}`;
}

function formatDate(ts) {

  return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
}

// ==================== 模态框 ====================
function showModal(config) {
  const container = document.getElementById('modal-container');
  const mask = document.createElement('div');
  mask.className = 'modal-mask';
  mask.onclick = (e) => {
    if (e.target === mask && config.closeOnMask !== false) closeModal();
  };

  let footerHtml = '';
  if (config.showFooter !== false) {
    footerHtml = `
      <div class="modal-footer">
        <button class="btn btn-default" id="modal-cancel-btn">${config.cancelText || '取消'}</button>
        <button class="btn ${config.confirmType || 'btn-primary'}" id="modal-confirm-btn">${config.confirmText || '确定'}</button>
      </div>
    `;
  }

  mask.innerHTML = `
    <div class="modal">
      <div class="modal-header">${config.title}</div>
      <div class="modal-body">${config.body}</div>
      ${footerHtml}
    </div>
  `;
  container.appendChild(mask);

  // Use addEventListener to preserve closure context
  if (config.showFooter !== false) {
    mask.querySelector('#modal-cancel-btn').addEventListener('click', () => {
      if (config.onCancel) config.onCancel();
      else closeModal();
    });
    mask.querySelector('#modal-confirm-btn').addEventListener('click', () => {
      if (config.onConfirm) config.onConfirm();
    });
  }

  if (config.onShow) config.onShow();
}

function closeModal() {
  const container = document.getElementById('modal-container');
  if (container.lastChild) container.removeChild(container.lastChild);
}

// 二次确认
function confirmDialog(config) {
  const { title, message, confirmText = '确认撤销', dangerText, onConfirm } = config;
  
  let body = `<p style="font-size:14px;color:#4a5568;line-height:1.6;">${message}</p>`;
  if (dangerText) {
    body += `
      <p style="font-size:13px;color:#e53e3e;margin-top:10px;">⚠️ 此操作不可撤销，请谨慎操作！</p>
      <input type="text" class="confirm-input" id="confirm-danger-input" placeholder="请输入「${dangerText}」确认">
    `;
  }
  
  showModal({
    title,
    body,
    confirmText: '确认',
    confirmType: 'btn-danger',
    onConfirm: () => {
      if (dangerText) {
        const input = document.getElementById('confirm-danger-input');
        if (input.value.trim() !== dangerText) {
          showToast(`请输入「${dangerText}」确认`);
          return;
        }
      }
      closeModal();
      onConfirm();
    }
  });
}

function renderFlowDetailModal() {
  const { productId, warehouseId, filterType, filterDate } = flowDetailContext;
  const p = getProductById(productId);
  if (!p) return;
  const w = getWarehouseById(warehouseId);
  
  let flows = db.stockFlow.filter(f => f.productId === productId && f.warehouseId === warehouseId);
  
  // 筛选类型
  if (filterType) {
    if (filterType === 'in') flows = flows.filter(f => f.type === 'in');
    else if (filterType === 'out') flows = flows.filter(f => f.type === 'out');
  }
  
  // 筛选日期
  if (filterDate) {
    const filterDateStr = formatDate(filterDate);
    flows = flows.filter(f => formatDate(f.createTime) === filterDateStr);
  }
  
  flows.sort((a, b) => b.createTime - a.createTime);
  
  let flowHtml = '';
  if (flows.length === 0) {
    flowHtml = '<div style="text-align:center;color:#a0aec0;padding:16px;">暂无流水记录</div>';
  } else {
    flowHtml = '<div class="table-container"><table class="data-table"><thead><tr><th>日期时间</th><th>类型</th><th>数量</th><th>结存</th><th>备注</th></tr></thead><tbody>';
    flows.forEach(f => {
      const isIn = f.type === 'in';

        'stockIn': '入库',
        'revokeStockIn': '撤回入库',
        'stockOut': '出库',
        'revokeStockOut': '撤回出库',
        'transferIn': '调拨入',
        'transferOut': '调拨出',
      }[f.orderType] || (isIn ? '入库' : '出库');
      
      flowHtml += `

          <td><span class="tag ${isIn ? 'tag-green' : 'tag-red'}">${typeText}</span></td>

          <td style="font-size:12px;">${f.remark || '-'}</td>
        </tr>
      `;
    });
    flowHtml += '</tbody></table></div>';
  }
  
  const qty = getStock(productId, warehouseId);
  const body = `
    <div style="margin-bottom:12px;">
      <h3 style="margin-bottom:4px;">${p.name} ${p.isCustom ? '<span class="tag tag-blue">定制</span>' : ''}</h3>

    </div>
    <div style="display:flex;gap:8px;margin-bottom:10px;flex-wrap:wrap;">
      <button class="btn btn-sm btn-primary" onclick="exportFlowExcel()">导出Excel</button>
      <button class="btn btn-sm btn-success" onclick="exportFlowImage()">导出图片</button>

      <select id="flow-filter-type" onchange="updateFlowFilter()">

function renderStockInList() {
  const warehouseId = document.getElementById('in-list-filter-warehouse').value;
  const status = document.getElementById('in-list-filter-status').value;
  const productSearch = (document.getElementById('in-list-filter-product') || {}).value || '';
  const dateStart = (document.getElementById('in-list-filter-date-start') || {}).value || '';
  const dateEnd = (document.getElementById('in-list-filter-date-end') || {}).value || '';
  const content = document.getElementById('stock-in-list-content');
  
  let orders = [...db.stockInOrders];
  if (warehouseId) orders = orders.filter(o => o.warehouseId === warehouseId);
  if (status) orders = orders.filter(o => o.status === status);
  if (productSearch) {
    const kw = productSearch.toLowerCase();
    orders = orders.filter(o => o.details.some(d => {
      const p = getProductById(d.productId);
      return p && p.name.toLowerCase().indexOf(kw) >= 0;
    }));
  }
  if (dateStart) {
    const startTime = new Date(dateStart + 'T00:00:00').getTime();
    orders = orders.filter(o => o.createTime >= startTime);
  }
  if (dateEnd) {
    const endTime = new Date(dateEnd + 'T23:59:59').getTime();
    orders = orders.filter(o => o.createTime <= endTime);
  }
  
  if (orders.length === 0) {
    content.innerHTML = '<div class="empty-state"><div class="empty-icon">📭</div>暂无入库记录</div>';
    return;

  let html = '<div class="card" style="padding:0;overflow:hidden;"><div class="table-container"><table class="data-table"><thead><tr><th>仓库</th><th>产品名称</th><th>数量</th><th>来源</th><th>日期时间</th><th>备注</th><th>操作</th></tr></thead><tbody>';
  orders.forEach(o => {
    const warehouse = getWarehouseById(o.warehouseId);
    const isRevoked = o.status === 'revoked';
    const rowClass = isRevoked ? 'revoked-row' : '';
    // 多个产品合并显示
    const productNames = o.details.map(d => {
      const p = getProductById(d.productId);
      return p ? p.name : '-';
    }).join('、');
    const totalQty = o.details.reduce((s, d) => s + d.qty, 0);
    const canRevoke = !isRevoked && (isAdmin() || o.createBy === currentUser.id);
    
    html += `<tr class="${rowClass}">`;
    html += `<td>${warehouse ? warehouse.name : '-'}</td>`;
    html += `<td>${productNames}</td>`;
    html += `<td>${totalQty}</td>`;
    html += `<td>${o.source}${isRevoked ? '（已撤销）' : ''}</td>`;
    html += `<td style="white-space:nowrap;font-size:12px;">${formatTime(o.createTime)}</td>`;
    html += `<td style="font-size:12px;max-width:120px;overflow:hidden;text-overflow:ellipsis;">${o.remark || '-'}</td>`;

function showStockInDetail(orderId) {
  const order = db.stockInOrders.find(o => o.id === orderId);
  if (!order) return;
  
  const warehouse = getWarehouseById(order.warehouseId);
  const isRevoked = order.status === 'revoked';
  const canRevoke = !isRevoked && (isAdmin() || order.createBy === currentUser.id);

  order.details.forEach(d => {

    detailsHtml += `

      </div>

  if (isRevoked) {
    revokeHtml = `
      <p style="margin-bottom:8px;"><strong>撤回原因：</strong>${order.revokeReason || '-'}</p>
      <p style="margin-bottom:8px;"><strong>撤回时间：</strong>${order.revokeTime ? formatTime(order.revokeTime) : '-'}</p>
      <p style="margin-bottom:8px;"><strong>撤回人：</strong>${order.revokeByName || '-'}</p>
    `;
  } else if (canRevoke) {
    revokeHtml = `
      <div style="margin-top:12px;border-top:1px solid #e2e8f0;padding-top:12px;">
        <div class="form-group">
          <label class="form-label">撤回原因</label>
          <textarea class="form-textarea" id="revoke-in-reason" placeholder="请输入撤回原因"></textarea>
        </div>
      </div>
    `;
  }
  
  const body = `
    <p style="margin-bottom:8px;"><strong>仓库：</strong>${warehouse ? warehouse.name : '-'}</p>
    <p style="margin-bottom:8px;"><strong>来源：</strong>${order.source}</p>
    <p style="margin-bottom:8px;"><strong>操作人：</strong>${order.createByName}</p>
    <p style="margin-bottom:8px;"><strong>时间：</strong>${formatTime(order.createTime)}</p>
    <p style="margin-bottom:8px;"><strong>备注：</strong>${order.remark || '-'}</p>
    <div style="margin-top:12px;border-top:1px solid #e2e8f0;padding-top:12px;">
      <p style="font-weight:600;margin-bottom:8px;">入库明细</p>

          return;
        }

      }
    });
  } else {
    showModal({
      title: '入库单详情',
      body,
      showFooter: false,
    });
  }
}

function revokeStockIn(orderId, reason) {
  const order = db.stockInOrders.find(o => o.id === orderId);
  if (!order || order.status !== 'normal') return;
  
  // 检查库存是否充足（撤销入库 = 扣减库存）
  for (const d of order.details) {
    const currentStock = getStock(d.productId, order.warehouseId);
    if (currentStock < d.qty) {
      const p = getProductById(d.productId);
      showToast(`撤销失败：${p ? p.name : '产品'}库存不足（当前${currentStock}台）`);
      return;
    }
  }
  
  confirmDialog({
    title: '确认撤回',
    message: `确定要撤回此入库单吗？撤回后库存将原路返回（反向扣减）。\n撤回原因：${reason}`,
    dangerText: '确认撤回',
    onConfirm: () => {
      const now = Date.now();
      order.status = 'revoked';
      order.revokeTime = now;
      order.revokeBy = currentUser.id;
      order.revokeByName = currentUser.displayName;
      order.revokeReason = reason;
      
      // 生成反向流水（库存原路返回=扣减）
      order.details.forEach(d => {
        const beforeQty = getStock(d.productId, order.warehouseId);
        const afterQty = beforeQty - d.qty;
        db.stockFlow.push({
          id: genId('F'),
          productId: d.productId,
          warehouseId: order.warehouseId,
          type: 'out',
          qty: d.qty,
          beforeQty,
          afterQty,
          orderId: order.id,
          orderType: 'revokeStockIn',
          remark: '撤回入库：' + reason,
          createBy: currentUser.id,
          createByName: currentUser.displayName,
          createTime: now,
        });
      });
      
      addOperationLog('撤回入库', `撤回入库单 ${orderId}，原因：${reason}`);
      saveData();
      showToast('已撤回');
      closeModal();
      renderStockInList();
    }
  });
}

function submitStockOut() {
  const warehouseId = document.getElementById('out-warehouse').value;
  const destination = document.getElementById('out-destination').value.trim();
  const channel = document.getElementById('out-channel').value;
  const remark = document.getElementById('out-remark').value.trim();
  
  if (!destination) {
    showToast('请填写出货去向');
    return;
  }
  
  const validDetails = outDetails.filter(d => d.productId && d.qty > 0);
  if (validDetails.length === 0) {
    showToast('请至少添加一条有效明细');
    return;
  }
  
  const warehouse = getWarehouseById(warehouseId);
  
  // 库存校验
  for (const d of validDetails) {
    const current = getStock(d.productId, warehouseId);
    if (d.qty > current) {
      const p = getProductById(d.productId);
      showToast(`库存不足：${p ? p.name : '产品'} 当前${current}台`);
      return;
    }
  }
  
  // 定制校验
  const customProducts = validDetails.filter(d => {
    const p = getProductById(d.productId);
    return p && p.isCustom;
  });
  
  // 定制产品从普通仓出库
  if (warehouse.type === 'normal' && customProducts.length > 0 && !isAdmin()) {
    showToast('定制产品不能从普通仓出库，请选择定制仓');
    return;
  }
  
  // 定制仓出货去向校验
  if (warehouse.type === 'custom' && warehouse.bindCustomer) {
    if (destination !== warehouse.bindCustomer && !isAdmin()) {
      showToast(`定制仓绑定客户为「${warehouse.bindCustomer}」，出货去向不一致`);
      return;
    }
  }
  
  // 如果是管理员且有定制校验问题，需要确认
  const needForceConfirm = isAdmin() && (
    (warehouse.type === 'normal' && customProducts.length > 0) ||
    (warehouse.type === 'custom' && warehouse.bindCustomer && destination !== warehouse.bindCustomer)
  );
  
  const doSubmit = () => {
    const btn = document.getElementById('submit-out-btn');
    btn.disabled = true;
    
    const now = Date.now();
    const orderId = genId('OUT');
    
    const order = {
      id: orderId,
      warehouseId,
      destination,
      channel,
      remark,
      details: validDetails,
      status: 'normal',
      createBy: currentUser.id,
      createByName: currentUser.displayName,
      createTime: now,
      forceOverride: needForceConfirm,
    };
    
    db.stockOutOrders.unshift(order);
    
    // 生成流水
    validDetails.forEach(d => {
      const beforeQty = getStock(d.productId, warehouseId);
      const afterQty = beforeQty - d.qty;
      db.stockFlow.push({
        id: genId('F'),
        productId: d.productId,
        warehouseId,
        type: 'out',
        qty: d.qty,
        beforeQty,
        afterQty,
        orderId,
        orderType: 'stockOut',
        remark: destination,
        createBy: currentUser.id,
        createByName: currentUser.displayName,
        createTime: now,
      });
    });
    
    addOperationLog('出库', `出库单 ${orderId}，${validDetails.length} 种产品，去向：${destination}`);
    saveData();
    showToast('出库成功');
    btn.disabled = false;
    
    setTimeout(() => goPage('stock-out-list'), 800);
  };
  
  if (needForceConfirm) {
    confirmDialog({
      title: '强制放行确认',
      message: '检测到定制产品出库规则不匹配，作为管理员确认强制放行？',
      dangerText: '确认强制放行',
      onConfirm: doSubmit
    });
  } else {
    doSubmit();
  }
}

function revokeStockOut(orderId, reason) {
  const order = db.stockOutOrders.find(o => o.id === orderId);
  if (!order || order.status !== 'normal') return;
  
  confirmDialog({
    title: '确认撤回',
    message: `确定要撤回此出库单吗？撤回后库存将原路返回（加回原仓库）。\n撤回原因：${reason}`,
    dangerText: '确认撤回',
    onConfirm: () => {
      const now = Date.now();
      order.status = 'revoked';
      order.revokeTime = now;
      order.revokeBy = currentUser.id;
      order.revokeByName = currentUser.displayName;
      order.revokeReason = reason;
      
      // 生成反向流水（库存原路返回=加回）
      order.details.forEach(d => {
        const beforeQty = getStock(d.productId, order.warehouseId);
        const afterQty = beforeQty + d.qty;
        db.stockFlow.push({
          id: genId('F'),
          productId: d.productId,
          warehouseId: order.warehouseId,
          type: 'in',
          qty: d.qty,
          beforeQty,
          afterQty,
          orderId: order.id,
          orderType: 'revokeStockOut',
          remark: '撤回出库：' + reason,
          createBy: currentUser.id,
          createByName: currentUser.displayName,
          createTime: now,
        });
      });
      
      addOperationLog('撤回出库', `撤回出库单 ${orderId}，原因：${reason}`);
      saveData();
      showToast('已撤回');
      closeModal();
      renderStockOutList();
    }
  });
}

// 产品档案导入CSV
function importProductCSV(event) {
  var file = event.target.files[0];
  if (!file) return;
  var reader = new FileReader();
  reader.onload = function(e) {
    try {
      var text = e.target.result;
      var lines = text.split(/\r?\n/).filter(function(l) { return l.trim().length > 0; });
      if (lines.length <= 1) { showToast('文件内容为空'); return; }
      
      var imported = 0;
      var updated = 0;
      for (var i = 1; i < lines.length; i++) {
        var cols = parseCSVLine(lines[i]);
        if (cols.length < 2) continue;
        var name = (cols[0] || '').trim();
        var code = (cols[1] || '').trim();
        var spec = (cols[2] || '').trim();
        var perBox = parseInt(cols[3]) || 1;
        var safetyStock = parseInt(cols[4]) || 0;
        var isCustom = cols[5] === '是' || cols[5] === 'true' || cols[5] === '1';
        
        if (!name) continue;
        
        // 按编码或名称匹配，存在则更新，不存在则新增
        var existing = db.products.find(function(p) {
          return (code && p.code === code) || p.name === name;
        });
        
        if (existing) {
          existing.name = name;
          if (code) existing.code = code;
          if (spec) existing.spec = spec;
          if (perBox > 0) existing.perBox = perBox;
          if (safetyStock >= 0) existing.safetyStock = safetyStock;
          existing.isCustom = isCustom;
          updated++;
        } else {
          db.products.push({
            id: genId('P'),
            name: name,
            code: code || ('P' + String(Date.now()).slice(-6) + i),
            spec: spec || '',
            perBox: perBox || 1,
            safetyStock: safetyStock || 0,
            isCustom: isCustom,
            createTime: Date.now(),
            createBy: currentUser.id,
          });
          imported++;
        }
      }
      
      addOperationLog('导入产品档案', '新增' + imported + '条，更新' + updated + '条');
      saveData();
      renderProductList();
      showToast('导入成功：新增' + imported + '条，更新' + updated + '条');
    } catch (err) {
      showToast('导入失败：' + err.message);
    }
    event.target.value = '';
  };
  reader.readAsText(file, 'UTF-8');
}

// 从文字中解析订单行
function ocrParseText(text) {
  var lines = text.split(/[\n\r]+/).filter(function(l) { return l.trim().length > 0; });
  var results = [];
  lines.forEach(function(line) {
    line = line.trim();
    // 尝试匹配"产品名 数量"、"产品名x数量"、"产品名：数量"等格式
    var match = line.match(/^(.+?)[\s\*xX×:：]+(\d+)/);
    if (match) {
      var name = match[1].replace(/[（）\(\)]/g, '').trim();
      var qty = parseInt(match[2]) || 0;
      if (name && qty > 0) {
        results.push({ ocrName: name, qty: qty });
      }
    } else {
      // 如果整行看起来像纯数字跳过
      if (!/^\d+$/.test(line) && line.length > 0) {
        results.push({ ocrName: line, qty: 1 });
      }
    }
  });
  return results;
}

// 模糊匹配产品 - 返回 {product, score}
function ocrMatchProduct(ocrName) {
  var matchedProduct = null;
  var bestScore = 0;
  var bestMatchedName = '';
  db.products.forEach(function(p) {
    var result = ocrFuzzyMatch(ocrName, p.name);
    if (result.score > bestScore) {
      bestScore = result.score;
      matchedProduct = p;
      bestMatchedName = result.matchedName;
    }
  });
  return { product: matchedProduct, score: bestScore, matchedName: bestMatchedName };
}

}

// 模糊匹配 - 综合相似度算法
// 返回 {score: 0~1, matchedName: 匹配到的产品名称}
function ocrFuzzyMatch(ocrText, productName) {
  if (!ocrText || !productName) return { score: 0, matchedName: '' };
  var o = ocrText.toLowerCase().trim();
  var p = productName.toLowerCase().trim();
  
  // 1. 完全匹配
  if (o === p) return { score: 1, matchedName: productName };
  
  var score = 0;
  
  // 2. 包含匹配
  if (p.indexOf(o) >= 0 || o.indexOf(p) >= 0) {
    // 包含关系，得分基于长度比
    var longStr = o.length > p.length ? o : p;
    var shortStr = o.length > p.length ? p : o;
    score = Math.max(score, shortStr.length / longStr.length * 0.85);
  }
  
  // 3. 编辑距离相似度 (Levenshtein-based)
  var distScore = 1 - levenshteinDistance(o, p) / Math.max(o.length, p.length);
  if (distScore > 0) score = Math.max(score, distScore);
  
  // 4. 分词重叠匹配 - 拆分关键词计算重叠率
  var oTokens = o.split(/[\s\-_,，。、（）()]+/).filter(function(t) { return t.length > 0; });
  var pTokens = p.split(/[\s\-_,，。、（）()]+/).filter(function(t) { return t.length > 0; });
  if (oTokens.length > 0 && pTokens.length > 0) {
    var matchCount = 0;
    oTokens.forEach(function(ot) {
      pTokens.forEach(function(pt) {
        if (ot === pt || ot.indexOf(pt) >= 0 || pt.indexOf(ot) >= 0) matchCount++;
      });
    });
    var tokenScore = matchCount / Math.max(oTokens.length, pTokens.length);
    score = Math.max(score, tokenScore * 0.8);
  }
  
  // 5. 部分字符匹配 - 识别名称中的关键字符在产品名中出现的比例
  if (o.length >= 2 && p.length >= 2) {
    var charMatch = 0;
    for (var ci = 0; ci < o.length; ci++) {
      if (p.indexOf(o.charAt(ci)) >= 0) charMatch++;
    }
    var charScore = charMatch / o.length;
    if (charScore > 0.5) score = Math.max(score, charScore * 0.6);
  }
  
  return { score: Math.min(score, 1), matchedName: productName };
}

// 编辑距离算法
function levenshteinDistance(s1, s2) {
  if (s1.length === 0) return s2.length;
  if (s2.length === 0) return s1.length;
  var matrix = [];
  for (var i = 0; i <= s2.length; i++) matrix[i] = [i];
  for (var j = 0; j <= s1.length; j++) matrix[0][j] = j;
  for (var ii = 1; ii <= s2.length; ii++) {
    for (var jj = 1; jj <= s1.length; jj++) {
      if (s2.charAt(ii - 1) === s1.charAt(jj - 1)) {
        matrix[ii][jj] = matrix[ii - 1][jj - 1];
      } else {
        matrix[ii][jj] = Math.min(matrix[ii - 1][jj - 1] + 1, matrix[ii - 1][jj] + 1, matrix[ii][jj - 1] + 1);
      }
    }
  }
  return matrix[s2.length][s1.length];
}

// 渲染识别结果
function ocrRenderResults() {
  var container = document.getElementById('ocr-result-table');
  var whId = ocrGetCurrentWhId();
  var totalQty = 0;
  var totalBoxes = 0;
  var totalRemain = 0;
  var selectedCount = 0;
  
  if (ocrResultList.length === 0) {
    container.innerHTML = '<div style="text-align:center;color:#a0aec0;padding:16px;">暂无识别结果</div>';
    document.getElementById('ocr-stats-bar').textContent = '共 0 种 · 总数量 0';
    return;
  }
  
  // 表格 - 每列都给足够宽度，整体可左右滑动
  var html = '<table class="data-table" style="min-width:700px;border-collapse:collapse;"><thead><tr style="background:#f0f7ff;">';
  html += '<th style="width:36px;padding:6px;text-align:center;border:1px solid #e2e8f0;">选</th>';
  html += '<th style="width:40px;padding:6px;text-align:center;border:1px solid #e2e8f0;">序号</th>';
  html += '<th style="min-width:140px;padding:6px;border:1px solid #e2e8f0;">识别名称</th>';
  html += '<th style="min-width:140px;padding:6px;border:1px solid #e2e8f0;">匹配名称</th>';
  html += '<th style="min-width:80px;padding:6px;border:1px solid #e2e8f0;">相似度</th>';
  html += '<th style="min-width:70px;padding:6px;border:1px solid #e2e8f0;">数量</th>';
  html += '<th style="min-width:80px;padding:6px;border:1px solid #e2e8f0;">装箱</th>';
  html += '<th style="min-width:100px;padding:6px;border:1px solid #e2e8f0;">库存校验</th>';
  html += '</tr></thead><tbody>';
  
  ocrResultList.forEach(function(r, i) {
    var product = r.productId ? getProductById(r.productId) : null;
    var packText = '-';
    if (product && product.perBox > 0) {
      var boxes = Math.floor(r.qty / product.perBox);
      var remain = r.qty % product.perBox;
      packText = boxes + '箱' + (remain > 0 ? remain + '台' : '');
      totalBoxes += boxes;
      totalRemain += remain;
    }
    
    var stockInfo = '<span style="color:#a0aec0;">未匹配</span>';
    if (product && whId) {
      var stock = getStock(product.id, whId);
      if (r.qty <= stock) {
        stockInfo = '<span style="color:#38a169;font-weight:600;">充足(' + stock + ')</span>';
      } else {
        stockInfo = '<span style="color:#e53e3e;font-weight:600;">缺货(' + stock + ',缺' + (r.qty - stock) + ')</span>';
      }
    } else if (product) {
      stockInfo = '<span style="color:#3182ce;font-weight:600;">可入库</span>';
    }
    
    totalQty += r.qty;
    if (r.selected) selectedCount++;
    
    // 相似度显示
    var scorePercent = Math.round((r.matchScore || 0) * 100);
    var scoreColor = scorePercent >= 80 ? '#38a169' : (scorePercent >= 50 ? '#d69e2e' : '#a0aec0');
    var scoreText = r.productId ? '<span style="color:' + scoreColor + ';font-weight:600;">' + scorePercent + '%</span>' : '<span style="color:#cbd5e0;">-</span>';
    
    // 匹配产品下拉 - 自动选中，显示完整产品名称
    var matchSelect = '<select style="width:100%;font-size:12px;padding:4px 6px;border:1px solid #cbd5e0;border-radius:6px;background:#fff;" onchange="ocrOnProductChange(' + i + ', this.value)"><option value="">未匹配</option>' +
      db.products.map(function(p) { return '<option value="' + p.id + '"' + (r.productId === p.id ? ' selected' : '') + '>' + escapeHtml(p.name) + '</option>'; }).join('') + '</select>';
    
    var rowStyle = r.selected ? '' : 'opacity:0.4;';
    var bgStyle = i % 2 === 0 ? 'background:#fff;' : 'background:#f7fafc;';
    html += '<tr style="' + rowStyle + bgStyle + '">';
    html += '<td style="padding:6px;text-align:center;border:1px solid #e2e8f0;"><input type="checkbox" ' + (r.selected ? 'checked' : '') + ' onchange="ocrResultList[' + i + '].selected=this.checked;ocrRenderResults();"></td>';
    html += '<td style="padding:6px;text-align:center;border:1px solid #e2e8f0;font-size:13px;color:#718096;">' + (i + 1) + '</td>';
    // 识别名称 - 纯文字显示，不截断，自动换行
    html += '<td style="padding:6px;border:1px solid #e2e8f0;font-size:13px;color:#2d3748;word-break:break-all;white-space:normal;line-height:1.4;">' + escapeHtml(r.ocrName) + '</td>';
    // 匹配名称 - 下拉框，自动选中
    html += '<td style="padding:4px 6px;border:1px solid #e2e8f0;">' + matchSelect + '</td>';
    html += '<td style="padding:6px;text-align:center;border:1px solid #e2e8f0;font-size:13px;white-space:nowrap;">' + scoreText + '</td>';
    // 数量 - 可编辑
    html += '<td style="padding:4px 6px;border:1px solid #e2e8f0;"><input type="number" value="' + r.qty + '" style="width:60px;border:1px solid #cbd5e0;border-radius:6px;padding:4px 6px;font-size:13px;text-align:center;" onchange="ocrResultList[' + i + '].qty=parseInt(this.value)||0;ocrRenderResults();"></td>';
    html += '<td style="padding:6px;border:1px solid #e2e8f0;font-size:12px;color:#718096;white-space:nowrap;">' + packText + '</td>';
    html += '<td style="padding:6px;border:1px solid #e2e8f0;font-size:12px;white-space:nowrap;">' + stockInfo + '</td>';
    html += '</tr>';
  });
  
  html += '</tbody></table>';
  container.innerHTML = html;
  
  // 更新统计栏
  var statsBar = document.getElementById('ocr-stats-bar');
  var statsText = '共 ' + ocrResultList.length + ' 种 · 总数量 ' + totalQty;
  if (totalBoxes > 0 || totalRemain > 0) {
    statsText += ' · ' + totalBoxes + '箱' + (totalRemain > 0 ? totalRemain + '台' : '');
  }
  if (selectedCount < ocrResultList.length) {
    statsText += ' · 已选' + selectedCount + '项';
  }
  statsBar.textContent = statsText;
}

// 当用户在结果中手动切换匹配产品时
function ocrOnProductChange(idx, productId) {
  ocrResultList[idx].productId = productId;
  if (productId) {
    var p = getProductById(productId);
    if (p) {
      // 同步更新相似度
      var result = ocrFuzzyMatch(ocrResultList[idx].ocrName, p.name);
      ocrResultList[idx].matchScore = result.score;
      ocrResultList[idx].matchedName = p.name;
    }
  } else {
    ocrResultList[idx].matchScore = 0;
    ocrResultList[idx].matchedName = '';
  }
  ocrRenderResults();
}

// HTML转义
function escapeHtml(text) {
  if (!text) return '';
  return text.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

// 一键出库 - 仅已勾选的，以产品档案名称为准
function ocrToOut() {
  var whId = ocrGetCurrentWhId();
  var valid = ocrResultList.filter(function(r) { return r.selected && r.productId && r.qty > 0; });
  if (valid.length === 0) { showToast('没有可出库的记录（请先勾选并匹配产品）'); return; }
  var canOut = valid.filter(function(r) {
    var p = getProductById(r.productId);
    if (!p || !whId) return false;
    return r.qty <= getStock(p.id, whId);
  });
  var shortItems = valid.filter(function(r) {
    var p = getProductById(r.productId);
    if (!p || !whId) return true;
    return r.qty > getStock(p.id, whId);
  });
  if (shortItems.length > 0) {
    var shortNames = shortItems.map(function(r) { var p = getProductById(r.productId); return p ? p.name : r.ocrName; });
    confirmDialog({
      title: '库存不足提示',
      message: shortItems.length + ' 项产品库存不足（' + shortNames.slice(0, 3).join('、') + '...），缺货项将无法出库。是否仅将充足项转入出库？\n充足项：' + canOut.length + ' 项',
      dangerText: '',
      onConfirm: function() { ocrDoTransferOut(canOut); }
    });
  } else {
    ocrDoTransferOut(valid);
  }
}

  }, 200);
}

function disableWarehouse(warehouseId) {
  const w = getWarehouseById(warehouseId);
  if (!w || w.isDefault) return;
  
  // 检查库存
  const stocks = getWarehouseStock(warehouseId);
  const totalQty = Object.values(stocks).reduce((a, b) => a + b, 0);
  
  if (totalQty !== 0) {
    let detail = '';
    Object.entries(stocks).forEach(([pid, qty]) => {
      if (qty !== 0) {
        const p = getProductById(pid);
        detail += `${p ? p.name : pid}: ${qty}台\n`;
      }
    });
    showModal({
      title: '无法停用',
      body: `<p style="color:#e53e3e;">仓库内还有库存，不能停用！</p>
        <div style="margin-top:10px;font-size:13px;background:#fff5f5;padding:10px;border-radius:6px;">
          ${detail.replace(/\n/g, '<br>')}
        </div>`,
      showFooter: true,
      confirmText: '知道了',
      confirmType: 'btn-default',
      onConfirm: closeModal
    });
    return;
  }
  
  // 定制仓有绑定客户确认
  const doDisable = () => {
    confirmDialog({
      title: '停用仓库',
      message: `确定要停用仓库「${w.name}」吗？停用后将不出现在出入库选项中。`,
      dangerText: '确认停用',
      onConfirm: () => {
        w.status = 'inactive';
        addOperationLog('停用仓库', w.name);
        saveData();
        renderWarehouseManage();
        showToast('已停用');
      }
    });
  };
  
  if (w.bindCustomer) {
    confirmDialog({
      title: '确认解绑',
      message: `该仓库绑定了客户「${w.bindCustomer}」，停用后绑定关系将保留，确认继续？`,
      dangerText: '',
      onConfirm: doDisable
    });
  } else {
    doDisable();
  }
}

function deleteWarehouse(warehouseId) {

  const stocks = getWarehouseStock(warehouseId);
  const totalQty = Object.values(stocks).reduce((a, b) => a + b, 0);

  if (totalQty !== 0) {
    let detail = '';
    Object.entries(stocks).forEach(([pid, qty]) => {
      if (qty !== 0) {
        const p = getProductById(pid);
        detail += `${p ? p.name : pid}: ${qty}台\n`;

    });
    showModal({
      title: '无法删除',

function doBackup() {
  const now = Date.now();
  const dateStr = formatDate(now).replace(/-/g, '');
  const timeStr = new Date(now).toTimeString().slice(0, 5).replace(':', '');
  const fileName = `阿平仓库管理_备份_${dateStr}_${timeStr}.json`;
  
  const backupData = JSON.stringify(db, null, 2);
  const blob = new Blob([backupData], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  
  var a = document.createElement('a');
  a.href = url;
  a.download = fileName;
  a.style.display = 'none';
  document.body.appendChild(a);
  setTimeout(function() {
    a.click();
    setTimeout(function() {
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }, 100);
  }, 0);
  
  db.backupLogs.unshift({
    id: genId('BKP'),
    fileName,
    status: '成功',
    storageLocation: '本地下载',
    userId: currentUser.id,
    userName: currentUser.displayName,
    createTime: now,
  });
  
  addOperationLog('数据备份', fileName);
  saveData();
  renderBackupPage();
  showToast('备份成功，文件已下载');
}

function handleRestoreFile(event) {
  const file = event.target.files[0];
  if (!file) return;
  
  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const data = JSON.parse(e.target.result);
      
      confirmDialog({
        title: '数据恢复确认',
        message: '恢复数据将覆盖当前所有数据，确定继续吗？',
        dangerText: '确认恢复',
        onConfirm: () => {
          db = data;
          saveData();
          addOperationLog('数据恢复', `从 ${file.name} 恢复`);
          showToast('恢复成功，请重新登录');
          setTimeout(logout, 1500);
        }
      });
    } catch (err) {
      showToast('文件格式错误');
    }
  };
  reader.readAsText(file);
  event.target.value = '';
}

          db.stockFlow = [];
          db.stockInOrders = [];
          db.stockOutOrders = [];
          break;
        case 'flow':
          db.stockFlow = [];
          break;
        case 'logs':
          db.operationLogs = [];
          db.backupLogs = [];
          break;
        case 'all':
          db.products = [];
          db.warehouses = [];
          db.stockFlow = [];
          db.stockInOrders = [];
          db.stockOutOrders = [];
          db.operationLogs = [];
          db.backupLogs = [];
          db.ocrLogs = [];
          db.transferOrders = [];
          break;
      }
      addOperationLog('清空数据', name);
      saveData();
      showToast('已清空' + name);
      renderBackupPage();
    }
  });
}

function submitTransfer() {
  const fromId = document.getElementById('transfer-from').value;
  const toId = document.getElementById('transfer-to').value;
  const productId = document.getElementById('transfer-product').value;
  const qty = parseInt(document.getElementById('transfer-qty').value);
  const remark = document.getElementById('transfer-remark').value.trim();
  
  if (fromId === toId) {
    showToast('调出仓和调入仓不能相同');
    return;
  }
  if (!productId || !qty || qty <= 0) {
    showToast('请选择产品并输入数量');
    return;
  }
  
  const currentStock = getStock(productId, fromId);
  if (qty > currentStock) {
    showToast('调出仓库存不足');
    return;
  }
  
  const fromWh = getWarehouseById(fromId);
  const toWh = getWarehouseById(toId);
  const p = getProductById(productId);
  
  // 定制产品跨类型调拨需要管理员
  const needForce = p && p.isCustom && fromWh.type === 'custom' && toWh.type === 'normal' && !isAdmin();
  if (needForce) {
    showToast('定制产品调拨到普通仓需管理员权限');
    return;
  }
  
  const needForce2 = p && !p.isCustom && fromWh.type === 'normal' && toWh.type === 'custom' && !isAdmin();
  if (needForce2) {
    showToast('普通产品调拨到定制仓需管理员权限');
    return;
  }
  
  const doTransfer = () => {
    const now = Date.now();
    const orderId = genId('TR');
    
    const order = {
      id: orderId,
      fromWarehouseId: fromId,
      toWarehouseId: toId,
      productId,
      qty,
      remark,
      status: 'normal',
      createBy: currentUser.id,
      createByName: currentUser.displayName,
      createTime: now,
    };
    
    db.transferOrders.unshift(order);
    
    // 调出流水（负向）
    const fromBefore = getStock(productId, fromId);
    db.stockFlow.push({
      id: genId('F'),
      productId,
      warehouseId: fromId,
      type: 'out',
      qty,
      beforeQty: fromBefore,
      afterQty: fromBefore - qty,
      orderId,
      orderType: 'transferOut',
      remark: `调拨至${toWh.name}`,
      createBy: currentUser.id,
      createByName: currentUser.displayName,
      createTime: now,
    });
    
    // 调入流水（正向）
    const toBefore = getStock(productId, toId);
    db.stockFlow.push({
      id: genId('F'),
      productId,
      warehouseId: toId,
      type: 'in',
      qty,
      beforeQty: toBefore,
      afterQty: toBefore + qty,
      orderId,
      orderType: 'transferIn',
      remark: `从${fromWh.name}调拨`,
      createBy: currentUser.id,
      createByName: currentUser.displayName,
      createTime: now,
    });
    
    addOperationLog('库存调拨', `${fromWh.name} → ${toWh.name}，${p.name} ${qty}台`);
    saveData();
    showToast('调拨成功');
    document.getElementById('transfer-qty').value = '';
    updateTransferStock();
    renderTransferList();
  };
  
  if ((needForce || needForce2) && isAdmin()) {
    confirmDialog({
      title: '强制调拨确认',
      message: '跨仓库类型调拨，作为管理员确认强制执行？',
      dangerText: '确认强制调拨',
      onConfirm: doTransfer
    });
  } else {
    doTransfer();
  }
}

// CSV导出工具
function exportCSV(filename, headers, rows) {
  var csv = [headers.join(',')];
  rows.forEach(function(r) {
    csv.push(r.map(function(c) { return '"' + String(c).replace(/"/g, '""') + '"'; }).join(','));
  });
  var blob = new Blob(['\ufeff' + csv.join('\n')], { type: 'text/csv;charset=utf-8' });
  var url = URL.createObjectURL(blob);
  var a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.style.display = 'none';
  document.body.appendChild(a);
  // 用setTimeout确保元素在DOM中后再触发
  setTimeout(function() {
    a.click();
    setTimeout(function() {
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }, 100);
  }, 0);
  showToast('已导出 ' + filename);
}

// 图片导出工具 - 将表格转为图片
function exportTableImage(filename, title, headers, rows) {
  const canvas = document.createElement('canvas');
  const ctx = canvas.getContext('2d');
  const colWidths = headers.map(() => 120);
  const rowHeight = 36;
  const headerHeight = 44;
  const padding = 20;
  const titleHeight = 50;
  
  const tableWidth = colWidths.reduce((a, b) => a + b, 0) + padding * 2;
  const tableHeight = titleHeight + headerHeight + rows.length * rowHeight + padding * 2;
  
  canvas.width = tableWidth;
  canvas.height = tableHeight;
  
  // 背景
  ctx.fillStyle = '#fff';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  // 标题
  ctx.fillStyle = '#1a365d';
  ctx.font = 'bold 18px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(title, canvas.width / 2, padding + 24);
  
  // 表头
  ctx.font = 'bold 13px sans-serif';
  ctx.textAlign = 'left';
  let x = padding;
  let y = padding + titleHeight;
  
  ctx.fillStyle = '#f7fafc';
  ctx.fillRect(x, y, colWidths.reduce((a, b) => a + b, 0), headerHeight);
  
  headers.forEach((h, i) => {
    ctx.fillStyle = '#4a5568';
    ctx.fillText(h, x + 8, y + 26);
    x += colWidths[i];
  });
  
  // 数据行
  y += headerHeight;
  rows.forEach((row, ri) => {
    x = padding;
    // 交替背景
    if (ri % 2 === 1) {
      ctx.fillStyle = '#f7fafc';
      ctx.fillRect(x, y, colWidths.reduce((a, b) => a + b, 0), rowHeight);
    }
    
    row.forEach((cell, ci) => {
      ctx.fillStyle = '#2d3748';
      ctx.font = '12px sans-serif';
      const text = String(cell).length > 10 ? String(cell).substring(0, 9) + '…' : String(cell);
      ctx.fillText(text, x + 8, y + 24);
      x += colWidths[ci];
    });
    y += rowHeight;
  });
  
  // 边框
  ctx.strokeStyle = '#e2e8f0';
  ctx.lineWidth = 1;
  ctx.strokeRect(padding, padding + titleHeight, colWidths.reduce((a, b) => a + b, 0), headerHeight + rows.length * rowHeight);
  
  var url = canvas.toDataURL('image/png');
  var a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.style.display = 'none';
  document.body.appendChild(a);
  setTimeout(function() {
    a.click();
    setTimeout(function() {
      document.body.removeChild(a);
    }, 100);
  }, 0);
  showToast('已导出 ' + filename);
}

  event.target.value = '';

// CSV行解析
function parseCSVLine(line) {
  const result = [];
  let current = '';
  let inQuotes = false;
  for (let i = 0; i < line.length; i++) {
    const c = line[i];
    if (c === '"') {
      if (inQuotes && line[i + 1] === '"') {
        current += '"';
        i++;
      } else {
        inQuotes = !inQuotes;
      }
    } else if (c === ',' && !inQuotes) {
      result.push(current);
      current = '';
    } else {
      current += c;
    }
  }
  result.push(current);
  return result;
}

// 入库清单导入CSV
function importInListCSV(event) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const text = e.target.result.replace(/^\ufeff/, '');
      const lines = text.split('\n').filter(l => l.trim());
      if (lines.length < 2) {
        showToast('文件无有效数据');
        return;
      }
      
      let imported = 0;
      for (let i = 1; i < lines.length; i++) {
        const cols = parseCSVLine(lines[i]);
        if (cols.length < 4) continue;
        
        const warehouseName = cols[0].trim();
        const productName = cols[1].trim();
        const qty = parseInt(cols[2]) || 0;
        const source = cols[3].trim();
        
        const product = db.products.find(p => p.name === productName);
        const warehouse = db.warehouses.find(w => w.name === warehouseName);
        
        if (product && warehouse && qty > 0) {
          const now = Date.now();
          const orderId = genId('IN');
          db.stockInOrders.push({
            id: orderId,
            warehouseId: warehouse.id,
            source: source || 'Excel导入',
            remark: '从Excel导入',
            details: [{ productId: product.id, qty }],
            status: 'normal',
            createBy: currentUser.id,
            createByName: currentUser.displayName,
            createTime: now,
          });
          const beforeQty = getStock(product.id, warehouse.id);
          db.stockFlow.push({
            id: genId('F'),
            productId: product.id,
            warehouseId: warehouse.id,
            type: 'in',
            qty,
            beforeQty,
            afterQty: beforeQty + qty,
            orderId,
            orderType: 'stockIn',
            remark: source || 'Excel导入',
            createBy: currentUser.id,
            createByName: currentUser.displayName,
            createTime: now,
          });
          imported++;
        }
      }
      
      addOperationLog('导入入库清单', `从Excel导入 ${imported} 条`);
      saveData();
      renderStockInList();
      showToast('导入成功 ' + imported + ' 条');
    } catch (err) {
      showToast('导入失败：' + err.message);
    }
  };
  reader.readAsText(file);
  event.target.value = '';
}

    try {
      const text = e.target.result.replace(/^\ufeff/, '');

        showToast('文件无有效数据');
        return;

        const cols = parseCSVLine(lines[i]);
        if (cols.length < 4) continue;

            createTime: now,
            forceOverride: false,
          });
          const beforeQty = getStock(product.id, warehouse.id);
          db.stockFlow.push({
            id: genId('F'),
            productId: product.id,
            warehouseId: warehouse.id,
            type: 'out',
            qty,
            beforeQty,
            afterQty: beforeQty - qty,
            orderId,
            orderType: 'stockOut',
            remark: destination || 'Excel导入',
            createBy: currentUser.id,
            createByName: currentUser.displayName,
